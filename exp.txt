    document.getElementById("myForm").addEventListener("submit", function(event) {

        var dept = document.getElementById("dept").value;
        var result = document.getElementById("result");

        if (name == "") {
            result.textContent = "Please enter your name.";
            result.style.color = "red";

            result.style.color = "red";

        else {
            result.textContent = "Welcome " + name +
            "! Age: " + age + ", Department: " + dept;
            result.style.color = "green";
        }
    });
</script>

1. Aim
To develop an AJAX-based web application that retrieves data from a server asynchronously without reloading the entire web page.

2. Algorithm
Start the program.
Create an HTML page with a button and a display area.
Create a JavaScript function named loadData().
Create an XMLHttpRequest object.
Open a connection to the data source using the open() method.
Set the request method as GET.
Send the request using the send() method.
Check whether the request is successfully completed.
Retrieve the response data from the server.
Display the data on the web page using DOM manipulation.
If an error occurs, display an error message.
Stop the program.

Program
```html
<!DOCTYPE html>
<html>
<head>
    <title>AJAX Web Application</title>
</head>
<body>

    <h2>AJAX Web Application</h2>

    <button onclick="loadData()">Load Data</button>

    <p id="result">Click the button to load data.</p>

    <script>
        function loadData() {

            var xhttp = new XMLHttpRequest();

            xhttp.onload = function() {

                if (this.status == 200) {
                    var data = JSON.parse(this.responseText);

                    document.getElementById("result").innerHTML =
                    "Title: " + data.title + "<br><br>" +
                    "Body: " + data.body;
                }
                else {
                    document.getElementById("result").innerHTML =
                    "Error loading data!";
                }
            };

            xhttp.open("GET",
            "https://jsonplaceholder.typicode.com/posts/1", true);

            xhttp.send();
        }
    </script>

</body>
</html>
```

Output:

Click the button to load data.

After clicking the Load Data button:

Body: quia et suscipit suscipit

      et cum reprehenderit molestiae
      ut ut quas totam nostrum rerum
      est autem sunt rem eveniet architecto

5. Result
Thus, an AJAX-based web application was successfully developed using JavaScript and XMLHttpRequest. Data was retrieved asynchronously from the server and displayed on the web page without reloading the entire page.

5. Server-Side Applications Using Node.js
1. Aim
To develop a simple server-side application using Node.js that creates a web server and responds to client requests.

2. Algorithm
Start the program.
Install Node.js in the system.
Create a JavaScript file named server.js.
Import the built-in HTTP module.
Create a server using the http.createServer() method.
Receive the client request and generate a response.
Set the response status code and content type.
Display a welcome message to the client.
Assign port number 3000 to the server.
Start the server using the listen() method.
Open a web browser and enter http://localhost:3000.
Display the server response.
Stop the program.

3. Program
```javascript
// Import HTTP module
const http = require("http");

// Create a server
const server = http.createServer((req, res) => {

    // Set response header
    res.writeHead(200, {
        "Content-Type": "text/html"
    });

    // Send response
    res.end(`
        <h1>Welcome to Node.js Server</h1>
        <p>This is a server-side application.</p>
    `);

});

// Start the server
server.listen(3000, () => {
    console.log("Server running at http://localhost:3000");
});
```

4. Execution

Step 1: Install Node.js in your system.
Step 2: Open Notepad or VS Code and type the program.
Step 3: Save the file as server.js.
Step 4: Open the terminal and execute the following command:
node server.js
Step 5: Open your web browser and enter:
http://localhost:3000

5. Output
Terminal Output:
Server running at http://localhost:3000

Browser Output:
+--------------------------------------+
|                                      |
|     Welcome to Node.js Server        |
|                                      |
|     This is a server-side            |
|     application.                     |
|                                      |
+--------------------------------------+

6. Result
Thus, a server-side application was successfully developed using Node.js. The Node.js HTTP server accepts client requests and sends an HTML response to the browser.

REST API Development and Database Connectivity
1. Aim
To develop a REST API using Node.js and Express.js and establish connectivity with a MySQL database to perform CRUD operations such as Create, Read, Update, and Delete.

2. Algorithm
Start the program.
Install Node.js and create a new project.
Install the required packages: Express, MySQL2, and CORS.
Import the required modules.
Create an Express application.
Connect to the MySQL database.
Create REST API routes:
GET – Retrieve student details.
POST – Insert new student details.
PUT – Update student details.
DELETE – Delete student details.
Receive client requests through API endpoints.
Execute the corresponding SQL queries.
Send the database result as a JSON response.
Start the server on port 3000.
Test the API using Postman.
Stop the program.

3. Program
Step 1: Install Required Packages
Open the terminal and execute:

npm init -y
npm install express mysql2 cors

Step 2: Create Database and Table
Open MySQL and execute:
CREATE DATABASE studentdb;
USE studentdb;
CREATE TABLE students (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100),
    email VARCHAR(100),
    department VARCHAR(100)
);

Step 3: Create server.js
The following is the complete simplified program.

```javascript
// Import modules
const express = require("express");
const mysql = require("mysql2");
const cors = require("cors");

// Create Express app
const app = express();

app.use(express.json());
app.use(cors());

// Connect to MySQL
const db = mysql.createConnection({
    host: "localhost",
    user: "root",
    password: "",
    database: "studentdb"
});

// Check database connection
db.connect((err) => {
    if (err) {
        console.log("Database connection failed");
    } else {
        console.log("Connected to MySQL database");
    }
});

// GET - Display students
app.get("/students", (req, res) => {
    db.query("SELECT * FROM students", (err, result) => {
        if (err) {
            res.send(err);
        } else {
            res.json(result);
        }
    });
});

// POST - Add student
app.post("/students", (req, res) => {
    const { name, email, department } = req.body;

    db.query(
        "INSERT INTO students (name, email, department) VALUES (?, ?, ?)",
        [name, email, department],
        (err, result) => {
            if (err) {
                res.send(err);
            } else {
                res.json({ message: "Student added successfully" });
            }
        }
    );
});

// PUT - Update student
app.put("/students/:id", (req, res) => {
    const { name, email, department } = req.body;
    const id = req.params.id;

    db.query(
        "UPDATE students SET name=?, email=?, department=? WHERE id=?",
        [name, email, department, id],
        (err, result) => {
            if (err) {
                res.send(err);
            } else {
                res.json({ message: "Student updated successfully" });
            }
        }
    );
});

// DELETE - Delete student
app.delete("/students/:id", (req, res) => {
    const id = req.params.id;

    db.query(
        "DELETE FROM students WHERE id=?",
        [id],
        (err, result) => {
            if (err) {
                res.send(err);
            } else {
                res.json({ message: "Student deleted successfully" });
            }
        }
    );
});

// Start server
app.listen(3000, () => {
    console.log("Server running at http://localhost:3000");
});
```
4. Execution
Step 1: Save the program as server.js.
Step 2: Start the MySQL server.
Step 3: Open the terminal and execute:
node server.js
Step 4: The following message will appear:
Connected to MySQL database
Server running at http://localhost:3000
